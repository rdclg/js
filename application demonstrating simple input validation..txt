Create a Servlet application demonstrating simple input validation.
Technology: Servlet validation
Where to write: src/java/ValidationServlet.java
Easy explanation: Validation checks that the name is not empty and that age is in a sensible range.
Code
import java.io.*;
import javax.servlet.*;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.*;
@WebServlet("/validate")
public class ValidationServlet extends HttpServlet {
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws IOException {
        response.setContentType("text/html;charset=UTF-8");
        PrintWriter out = response.getWriter();
        out.println("<h2>Simple Form</h2>");
        out.println("<form method='post'>");
        out.println("<label>Name: <input type='text' name='name' required></label><br>");
        out.println("<label>Age: <input type='number' name='age' required></label><br>");
        out.println("<button type='submit'>Submit</button></form>");
    }
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws IOException {
        response.setContentType("text/html;charset=UTF-8");
        PrintWriter out = response.getWriter();
        String name = request.getParameter("name").trim();
        String ageText = request.getParameter("age");
        if (name.isEmpty()) { out.println("Name is required"); return; }
        try {
            int age = Integer.parseInt(ageText);
            if (age < 1 || age > 120) out.println("Enter a valid age");
            else out.println("Input accepted for " + name);
        } catch (NumberFormatException e) { out.println("Age must be a number"); }
    }
}
Sample output: Input: name=Riya, age=14 -> Input accepted for Riya
How to run: Open `/validate` and try valid and invalid values.
