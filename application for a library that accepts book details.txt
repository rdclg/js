Create a Servlet application for a library that accepts book details and
displays the submitted information on a separate page.
Technology: Servlet + JSP
Where to write: `src/java/BookEntryServlet.java` and `Web Pages/bookdetails.jsp`
Easy explanation: The servlet accepts book details and forwards them to a separate JSP page using request attributes.
Code
// File: BookEntryServlet.java
import java.io.*;
import javax.servlet.*;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.*;
@WebServlet("/bookentry")
public class BookEntryServlet extends HttpServlet {
    protected void doGet(HttpServletRequest req, HttpServletResponse res) throws IOException {
        res.setContentType("text/html");
        res.getWriter().println("<form method='post'>Title: <input name='title'> Author: <input name='author'><button>Submit</button></form>");
    }
    protected void doPost(HttpServletRequest req, HttpServletResponse res) throws ServletException, IOException {
        req.setAttribute("title", req.getParameter("title"));
        req.setAttribute("author", req.getParameter("author"));
        req.getRequestDispatcher("/bookdetails.jsp").forward(req, res);
